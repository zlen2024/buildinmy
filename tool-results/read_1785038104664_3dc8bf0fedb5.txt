     2→     2→

     4→     4→import { motion, AnimatePresence } from "framer-motion";
     5→     5→import { createMalaysiaMap } from "krackedmaps";
     6→     6→import "krackedmaps/css";

     8→     8→import type { LocationPin } from "@/lib/map-store";
     9→     9→import { MapPinTooltip } from "@/components/MapPinTooltip";
    10→    10→import { QuickStatsOverlay } from "@/components/QuickStatsOverlay";
    11→    11→import { WeatherWidget } from "@/components/WeatherWidget";
    12→    12→import { StarfieldBackground } from "@/components/StarfieldBackground";
    13→    13→import { DistrictPanel } from "@/components/DistrictPanel";
    14→    14→import { Wifi, Activity, ZoomIn, ZoomOut, Crosshair, Plus, Minus, LocateFixed, ChevronDown } from "lucide-react";
    15→    15→import { cn } from "@/lib/utils";
    16→    16→
    17→    17→interface StateSelectPayload {
    18→    18→  state?: string;
    19→    19→  district?: string;
    20→    20→  code?: string;
    21→    21→  type?: string;
    22→    22→  slug?: string;

    26→    26→
    27→    27→interface DrillPayload {
    28→    28→  state?: string;
    29→    29→  district?: string;

    31→    31→  name?: string;

    33→    33→}
    34→    34→
    35→    35→export function MalaysiaMap() {
    36→    36→  const mapRef = useRef<HTMLDivElement>(null);
    37→    37→  const mapInstanceRef = useRef<ReturnType<typeof createMalaysiaMap> | null>(null);
    38→    38→  const pinsRef = useRef<Map<string, string>>(new Map());
    39→    39→
    40→    40→  const locations = useMapStore((s) => s.locations);
    41→    41→  const selectedState = useMapStore((s) => s.selectedState);
    42→    42→  const selectedDistrict = useMapStore((s) => s.selectedDistrict);
    43→    43→  const selectedVenue = useMapStore((s) => s.selectedVenue);
    44→    44→  const setSelectedState = useMapStore((s) => s.setSelectedState);
    45→    45→  const setSelectedDistrict = useMapStore((s) => s.setSelectedDistrict);
    46→    46→  const setSelectedVenue = useMapStore((s) => s.setSelectedVenue);
    47→    47→  const sidebarOpen = useMapStore((s) => s.sidebarOpen);
    48→    48→  const showWifiHeatmap = useMapStore((s) => s.showWifiHeatmap);
    49→    49→  const toggleWifiHeatmap = useMapStore((s) => s.toggleWifiHeatmap);
    50→    50→  const resetFilters = useMapStore((s) => s.resetFilters);
    51→    51→
    52→    52→  const [mapReady, setMapReady] = useState(false);
    53→    53→  const [hoveredVenue, setHoveredVenue] = useState<LocationPin | null>(null);
    54→    54→  const [tooltipPosition, setTooltipPosition] = useState<{ x: number; y: number } | null>(null);
    55→    55→  const [legendCollapsed, setLegendCollapsed] = useState(false);
    56→    56→
    57→    57→  // Custom dark theme for the map (uses krackedmaps' actual CSS variable names)
    58→    58→  const customTheme: Record<string, string> = {
    59→    59→    "--sea": "#06080f",
    60→    60→    "--grid": "rgba(224, 201, 127, 0.04)",
    61→    61→    "--grid-size": "44px",
    62→    62→    "--land": "#0f1830",
    63→    63→    "--land-hover": "#1a2548",
    64→    64→    "--land-active": "#e0c97f",
    65→    65→    "--stroke": "rgba(224, 201, 127, 0.45)",
    66→    66→    "--stroke-w": "1",
    67→    67→    "--district": "#0d1b3a",
    68→    68→    "--carve": "rgba(224, 201, 127, 0.35)",
    69→    69→    "--carve-w": "0.6",
    70→    70→    "--district-hi": "#e0c97f",
    71→    71→    "--label": "#e0c97f",
    72→    72→    "--label-dim": "rgba(224, 201, 127, 0.5)",
    73→    73→    "--pin": "#e94560",
    74→    74→    "--pin-ring": "rgba(233, 69, 96, 0.45)",
    75→    75→    "--accent": "#e0c97f",
    76→    76→    "--panel-bg": "#0d1b2a",
    77→    77→    "--panel-edge": "rgba(224, 201, 127, 0.25)",
    78→    78→  };
    79→    79→
    80→    80→  // Track whether we've drilled into a state (district view active)
    81→    81→  // Derived from selectedState + drill event flag for more accuracy
    82→    82→  const [drillEventFired, setDrillEventFired] = useState(false);
    83→    83→
    84→    84→  // Initialize map
    85→    85→  useEffect(() => {
    86→    86→    if (!mapRef.current) return;
    87→    87→
    88→    88→    const map = createMalaysiaMap(mapRef.current, {
    89→    89→      theme: customTheme,
    90→    90→      panel: false,
    91→    91→      tooltip: true,
    92→    92→      showDistricts: true,
    93→    93→      interactive: true,
    94→    94→      labels: false,
    95→    95→    });
    96→    96→
    97→    97→    mapInstanceRef.current = map;
    98→    98→    queueMicrotask(() => setMapReady(true));
    99→    99→
   100→   100→    // Handle state selection
   101→   101→    map.on("select", (payload: StateSelectPayload) => {
   102→   102→      if (payload.slug) {
   103→   103→        setSelectedState(payload.slug);
   104→   104→      } else {
   105→   105→        setSelectedState(null);
   106→   106→      }
   107→   107→    });
   108→   108→
   109→   109→    // Handle drill into state (zoom to district view).
   110→   110→    // krackedmaps emits `drill` with payload = state slug string (when drilling in)
   111→   111→    // or null (when drilling out). We also support object payload for forward-compat.
   112→   112→    map.on("drill", (payload: string | DrillPayload | null) => {
   113→   113→      const stateSlug = typeof payload === "string"
   114→   114→        ? payload
   115→   115→        : (payload && (payload.state || payload.slug)) || null;
   116→   116→      setDrillEventFired(!!stateSlug);
   117→   117→      if (!stateSlug) {
   118→   118→        // Drill-out event — clear district selection
   119→   119→        setSelectedDistrict(null);
   120→   120→      }
   121→   121→    });
   122→   122→
   123→   123→    return () => {
   124→   124→      if (map && typeof map.destroy === "function") {
   125→   125→        map.destroy();
   126→   126→      }
   127→   127→    };
   128→   128→  }, []);
   129→   129→
   130→   130→  // Build a lookup map from venue ID → venue object (stable, avoids stale closures)
   131→   131→  const venueById = useMemo(() => {
   132→   132→    const map = new Map<string, LocationPin>();
   133→   133→    for (const loc of locations) {
   134→   134→      map.set(loc.id, loc);
   135→   135→    }
   136→   136→    return map;
   137→   137→  }, [locations]);
   138→   138→
   139→   139→  // Get locations for selected state (+ district)
   140→   140→  const visibleLocations = useMemo(() => {
   141→   141→    if (!selectedState) return locations;
   142→   142→    const mappedState = SLUG_TO_STATE[selectedState] || selectedState;
   143→   143→    return locations.filter((loc) => {
   144→   144→      if (loc.state !== mappedState) return false;
   145→   145→      if (selectedDistrict && loc.district !== selectedDistrict) return false;
   146→   146→      return true;
   147→   147→    });
   148→   148→  }, [locations, selectedState, selectedDistrict]);
   149→   149→
   150→   150→  // Determine label visibility based on density and zoom level.
   151→   151→  // - No state selected + few pins (<=4): show full names
   152→   152→  // - Otherwise: HIDE labels — too dense or too zoomed in, use hover tooltips only
   153→   153→  // This keeps the map clean and avoids overlap; the MapPinTooltip + bottom-left
   154→   154→  // coordinate display provide context on hover.
   155→   155→  const labelMode: 'full' | 'hidden' =
   156→   156→    !selectedState && visibleLocations.length <= 4 ? 'full' : 'hidden';
   157→   157→
   158→   158→  // Label resolver based on mode
   159→   159→  const getPinLabel = useCallback(
   160→   160→    (loc: LocationPin): string => {
   161→   161→      switch (labelMode) {
   162→   162→        case 'full':
   163→   163→          return loc.name.length <= 24 ? loc.name : loc.name.slice(0, 24) + '…';
   164→   164→        case 'hidden':
   165→   165→        default:
   166→   166→          return ''; // No label — pins show on hover only
   167→   167→      }
   168→   168→    },
   169→   169→    [labelMode],
   170→   170→  );
   171→   171→
   172→   172→  // Update pins when locations or state changes
   173→   173→  const updatePins = useCallback(() => {
   174→   174→    if (!mapInstanceRef.current || !mapReady) return;
   175→   175→    const map = mapInstanceRef.current;
   176→   176→
   177→   177→    // Clear existing pins
   178→   178→    pinsRef.current.forEach((pinId) => {
   179→   179→      try { map.removePin(pinId); } catch { /* ignore */ }
   180→   180→    });
   181→   181→    pinsRef.current.clear();
   182→   182→
   183→   183→    // Add filtered location pins with label optimization
   184→   184→    for (const loc of visibleLocations) {
   185→   185→      try {
   186→   186→        const pinId = map.addPin({
   187→   187→          lng: loc.longitude,
   188→   188→          lat: loc.latitude,
   189→   189→          label: getPinLabel(loc),
   190→   190→          id: loc.id,
   191→   191→        });
   192→   192→        pinsRef.current.set(loc.id, pinId);
   193→   193→      } catch {
   194→   194→        // Skip pins outside bounds
   195→   195→      }
   196→   196→    }
   197→   197→  }, [visibleLocations, mapReady, getPinLabel]);
   198→   198→
   199→   199→  useEffect(() => {
   200→   200→    updatePins();
   201→   201→  }, [updatePins]);
   202→   202→
   203→   203→  // Pin click handler — krackedmaps has no pinClick event, so we use DOM delegation
   204→   204→  // Pins are rendered as <g class="pin" data-id="venueId"> in the SVG
   205→   205→  useEffect(() => {
   206→   206→    if (!mapInstanceRef.current || !mapReady) return;
   207→   207→
   208→   208→    const handlePinClick = (e: MouseEvent) => {
   209→   209→      const pinEl = (e.target as Element).closest('.pin');
   210→   210→      if (!pinEl) return;
   211→   211→
   212→   212→      const venueId = pinEl.getAttribute('data-id');
   213→   213→      if (!venueId) return;
   214→   214→
   215→   215→      const venue = venueById.get(venueId);
   216→   216→      if (venue) {
   217→   217→        setSelectedVenue(venue);
   218→   218→      }
   219→   219→    };
   220→   220→
   221→   221→    mapInstanceRef.current.root.addEventListener('click', handlePinClick);
   222→   222→    return () => {
   223→   223→      mapInstanceRef.current?.root.removeEventListener('click', handlePinClick);
   224→   224→    };
   225→   225→  }, [mapReady, venueById, setSelectedVenue]);
   226→   226→
   227→   227→  // Derived: are we drilled into a state?
   228→   228→  const isDrilledIn = !!selectedState && drillEventFired;
   229→   229→
   230→   230→  // Focus + drill into state when selected (district-level zoom)
   231→   231→  useEffect(() => {
   232→   232→    if (!mapInstanceRef.current) return;
   233→   233→    const map = mapInstanceRef.current;
   234→   234→    if (selectedState) {
   235→   235→      try {
   236→   236→        map.focus(selectedState);
   237→   237→        // Drill into the state to reveal district boundaries
   238→   238→        if (typeof map.drillInto === "function") {
   239→   239→          map.drillInto(selectedState);
   240→   240→        }
   241→   241→      } catch {
   242→   242→        // Focus might fail for states not in the main map
   243→   243→      }
   244→   244→    } else {
   245→   245→      // Exit district view
   246→   246→      try {
   247→   247→        if (typeof map.drillInto === "function") {
   248→   248→          map.drillInto(null);
   249→   249→        }
   250→   250→        map.focus(null);
   251→   251→      } catch {
   252→   252→        /* ignore */
   253→   253→      }
   254→   254→    }
   255→   255→  }, [selectedState]);
   256→   256→
   257→   257→  // Highlight selected district on the map
   258→   258→  useEffect(() => {
   259→   259→    if (!mapInstanceRef.current) return;
   260→   260→    const map = mapInstanceRef.current;
   261→   261→    try {
   262→   262→      if (selectedDistrict && typeof map.selectDistrict === "function") {
   263→   263→        map.selectDistrict(selectedDistrict);
   264→   264→      } else if (typeof map.selectDistrict === "function") {
   265→   265→        // Deselect district when null — call with empty string or skip
   266→   266→        // krackedmaps selectDistrict(key) — passing empty string clears highlight
   267→   267→        map.selectDistrict("");
   268→   268→      }
   269→   269→    } catch {
   270→   270→      /* ignore */
   271→   271→    }
   272→   272→  }, [selectedDistrict, isDrilledIn]);
   273→   273→
   274→   274→  // Pin hover tooltip — event delegation on SVG .pin[data-id] groups
   275→   275→  useEffect(() => {
   276→   276→    if (!mapInstanceRef.current || !mapReady) return;
   277→   277→
   278→   278→    const map = mapInstanceRef.current;
   279→   279→
   280→   280→    const handleMouseMove = (e: MouseEvent) => {
   281→   281→      const pinEl = (e.target as Element).closest('.pin[data-id]');
   282→   282→      if (!pinEl) return;
   283→   283→
   284→   284→      const venueId = pinEl.getAttribute('data-id');
   285→   285→      if (!venueId) return;
   286→   286→
   287→   287→      const venue = venueById.get(venueId);
   288→   288→      if (venue) {
   289→   289→        setHoveredVenue(venue);
   290→   290→        setTooltipPosition({ x: e.clientX, y: e.clientY });
   291→   291→      }
   292→   292→    };
   293→   293→
   294→   294→    const handleMouseLeave = () => {
   295→   295→      setHoveredVenue(null);
   296→   296→      setTooltipPosition(null);
   297→   297→    };
   298→   298→
   299→   299→    map.root.addEventListener('mousemove', handleMouseMove);
   300→   300→    map.root.addEventListener('mouseleave', handleMouseLeave);
   301→   301→    return () => {
   302→   302→      map.root.removeEventListener('mousemove', handleMouseMove);
   303→   303→      map.root.removeEventListener('mouseleave', handleMouseLeave);
   304→   304→    };
   305→   305→  }, [mapReady, venueById]);
   306→   306→
   307→   307→  // Venue count for selected state
   308→   308→  const venueCount = visibleLocations.length;
   309→   309→
   310→   310→  // Wi-Fi speed by state for heatmap
   311→   311→  const wifiSpeedByState = useMemo(() => {
   312→   312→    const stateMap = new Map<string, { total: number; count: number }>();
   313→   313→    for (const loc of locations) {
   314→   314→      const existing = stateMap.get(loc.state) || { total: 0, count: 0 };
   315→   315→      stateMap.set(loc.state, { total: existing.total + loc.avgDownloadMbps, count: existing.count + 1 });
   316→   316→    }
   317→   317→    return Array.from(stateMap.entries())
   318→   318→      .map(([state, { total, count }]) => ({
   319→   319→        state,
   320→   320→        avgSpeed: Math.round(total / count),
   321→   321→        count,
   322→   322→      }))
   323→   323→      .sort((a, b) => b.avgSpeed - a.avgSpeed);
   324→   324→  }, [locations]);
   325→   325→
   326→   326→  return (
   327→   327→    <div className="relative w-full h-full min-h-screen bg-[#0a0a0f] overflow-hidden">
   328→   328→      <StarfieldBackground />
   329→   329→      <div ref={mapRef} className="w-full h-full min-h-screen" />
   330→   330→
   331→   331→      {/* Vignette effect around map edges */}
   332→   332→      <div className="vignette-overlay" />
   333→   333→
   334→   334→      {/* Map pin hover tooltip — hidden when venue drawer is open */}
   335→   335→      <MapPinTooltip venue={selectedVenue ? null : hoveredVenue} position={selectedVenue ? null : tooltipPosition} />
   336→   336→
   337→   337→      {/* District panel — shown when state is selected */}
   338→   338→      <DistrictPanel />
   339→   339→
   340→   340→      {/* Drill controls — visible only when drilled in */}
   341→   341→      <AnimatePresence>
   342→   342→        {isDrilledIn && selectedState && (
   343→   343→          <motion.div
   344→   344→            initial={{ opacity: 0, y: -8, scale: 0.95 }}
   345→   345→            animate={{ opacity: 1, y: 0, scale: 1 }}
   346→   346→            exit={{ opacity: 0, y: -8, scale: 0.95 }}
   347→   347→            transition={{ type: "spring", damping: 25, stiffness: 300 }}
   348→   348→            className="absolute top-20 left-1/2 -translate-x-1/2 z-30 pointer-events-auto"
   349→   349→            style={{ marginLeft: "130px" }}
   350→   350→          >
   351→   351→            <div className="flex items-center gap-1.5 bg-[#0d1b2a]/85 backdrop-blur-md border border-[#e0c97f]/30 rounded-full p-1 shadow-lg state-badge-glow">
   352→   352→              <button
   353→   353→                onClick={() => {
   354→   354→                  setSelectedDistrict(null);
   355→   355→                  setSelectedState(null);
   356→   356→                }}
   357→   357→                aria-label="Exit district zoom"
   358→   358→                className="flex items-center gap-1.5 px-3 py-1.5 rounded-full text-[11px] font-medium text-[#e0c97f]/70 hover:text-[#e0c97f] hover:bg-[#e0c97f]/10 transition-all"
   359→   359→              >
   360→   360→                <ZoomOut className="w-3.5 h-3.5" />
   361→   361→                Exit zoom
   362→   362→              </button>
   363→   363→              {selectedDistrict && (
   364→   364→                <>
   365→   365→                  <div className="w-px h-4 bg-[#e0c97f]/15" />
   366→   366→                  <button
   367→   367→                    onClick={() => setSelectedDistrict(null)}
   368→   368→                    aria-label="Show all districts"
   369→   369→                    className="flex items-center gap-1.5 px-3 py-1.5 rounded-full text-[11px] font-medium text-[#e0c97f]/70 hover:text-[#e0c97f] hover:bg-[#e0c97f]/10 transition-all"
   370→   370→                  >
   371→   371→                    <Crosshair className="w-3.5 h-3.5" />
   372→   372→                    All districts
   373→   373→                  </button>
   374→   374→                </>
   375→   375→              )}
   376→   376→              <div className="w-px h-4 bg-[#e0c97f]/15" />
   377→   377→              <div className="flex items-center gap-1.5 px-3 py-1.5 text-[11px] text-[#e0c97f]/40">
   378→   378→                <ZoomIn className="w-3.5 h-3.5" />
   379→   379→                <span className="font-mono">
   380→   380→                  {selectedDistrict || "District view"}
   381→   381→                </span>
   382→   382→              </div>
   383→   383→            </div>
   384→   384→          </motion.div>
   385→   385→        )}
   386→   386→      </AnimatePresence>
   387→   387→
   388→   388→      {/* Map overlay gradient for readability */}
   389→   389→      <div className="absolute inset-0 pointer-events-none">
   390→   390→        <div className="absolute top-0 left-0 w-full h-32 bg-gradient-to-b from-[#0a0a0f]/80 to-transparent" />
   391→   391→        <div className="absolute bottom-0 left-0 w-full h-48 bg-gradient-to-t from-[#0a0a0f]/90 to-transparent" />
   392→   392→      </div>
   393→   393→
   394→   394→      {/* Weather Widget */}
   395→   395→      <WeatherWidget />
   396→   396→
   397→   397→      {/* Quick Stats Overlay */}
   398→   398→      <QuickStatsOverlay
   399→   399→        visibleLocations={visibleLocations}
   400→   400→        selectedState={selectedState}
   401→   401→        sidebarOpen={sidebarOpen}
   402→   402→      />
   403→   403→
   404→   404→      {/* Floating legend — glass-card enhanced with hover animations + collapse toggle */}
   405→   405→      <div className="absolute top-20 right-4 glass-card pointer-events-auto min-w-[140px] overflow-hidden">
   406→   406→        <div className="flex items-center justify-between px-4 pt-3.5 pb-2">
   407→   407→          <p className="text-[10px] text-[#e0c97f]/50 font-semibold uppercase tracking-widest">Legend</p>
   408→   408→          <button
   409→   409→            onClick={() => setLegendCollapsed((v) => !v)}
   410→   410→            aria-label={legendCollapsed ? "Expand legend" : "Collapse legend"}
   411→   411→            title={legendCollapsed ? "Expand" : "Collapse"}
   412→   412→            className="w-5 h-5 rounded flex items-center justify-center text-[#e0c97f]/40 hover:text-[#e0c97f] hover:bg-[#e0c97f]/8 transition-colors"
   413→   413→          >
   414→   414→            <motion.div animate={{ rotate: legendCollapsed ? -90 : 0 }} transition={{ duration: 0.2 }}>
   415→   415→              <ChevronDown className="w-3 h-3" />
   416→   416→            </motion.div>
   417→   417→          </button>
   418→   418→        </div>
   419→   419→        <AnimatePresence initial={false}>
   420→   420→          {!legendCollapsed && (
   421→   421→            <motion.div
   422→   422→              initial={{ height: 0, opacity: 0 }}
   423→   423→              animate={{ height: 'auto', opacity: 1 }}
   424→   424→              exit={{ height: 0, opacity: 0 }}
   425→   425→              transition={{ duration: 0.25, ease: [0.22, 1, 0.36, 1] }}
   426→   426→              className="overflow-hidden"
   427→   427→            >
   428→   428→              <div className="px-4 pb-4">
   429→   429→                <div className="space-y-2">
   430→   430→                  {(Object.entries(CATEGORY_CONFIG) as [string, { label: string; color: string; emoji: string }][]).map(([key, cfg]) => (
   431→   431→                    <div
   432→   432→                      key={key}
   433→   433→                      className="flex items-center gap-2.5 min-w-0 legend-item-hover cursor-default"
   434→   434→                      style={{ '--legend-color': cfg.color + '30' } as React.CSSProperties}
   435→   435→                    >
   436→   436→                      <div
   437→   437→                        className="w-5 h-5 rounded flex-shrink-0 flex items-center justify-center text-xs legend-icon"
   438→   438→                        style={{ backgroundColor: cfg.color + '20' }}
   439→   439→                      >
   440→   440→                        {cfg.emoji}
   441→   441→                      </div>
   442→   442→                      <span className="text-[11px] text-[#e0c97f]/70 whitespace-nowrap truncate">{cfg.label}</span>
   443→   443→                    </div>
   444→   444→                  ))}
   445→   445→                </div>
   446→   446→
