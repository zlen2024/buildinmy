     1→"use client";
     2→
     3→import { useEffect, useRef, useCallback, useState, useMemo } from "react";
     4→import { motion, AnimatePresence } from "framer-motion";
     5→import { createMalaysiaMap } from "krackedmaps";
     6→import "krackedmaps/css";
     7→import { useMapStore, STATE_DISPLAY_NAMES, CATEGORY_CONFIG, SLUG_TO_STATE } from "@/lib/map-store";
     8→import type { LocationPin } from "@/lib/map-store";
     9→import { MapPinTooltip } from "@/components/MapPinTooltip";
    10→import { QuickStatsOverlay } from "@/components/QuickStatsOverlay";
    11→import { WeatherWidget } from "@/components/WeatherWidget";
    12→import { StarfieldBackground } from "@/components/StarfieldBackground";
    13→import { DistrictPanel } from "@/components/DistrictPanel";
    14→import { Wifi, Activity, ZoomIn, ZoomOut, Crosshair, Plus, Minus, LocateFixed, ChevronDown } from "lucide-react";
    15→import { cn } from "@/lib/utils";
    16→
    17→interface StateSelectPayload {
    18→  state?: string;
    19→  district?: string;
    20→  code?: string;
    21→  type?: string;
    22→  slug?: string;
    23→  name?: string;
    24→  [key: string]: unknown;
    25→}
    26→
    27→interface DrillPayload {
    28→  state?: string;
    29→  district?: string;
    30→  slug?: string;
    31→  name?: string;
    32→  [key: string]: unknown;
    33→}
    34→
    35→export function MalaysiaMap() {
    36→  const mapRef = useRef<HTMLDivElement>(null);
    37→  const mapInstanceRef = useRef<ReturnType<typeof createMalaysiaMap> | null>(null);
    38→  const pinsRef = useRef<Map<string, string>>(new Map());
    39→
    40→  const locations = useMapStore((s) => s.locations);
    41→  const selectedState = useMapStore((s) => s.selectedState);
    42→  const selectedDistrict = useMapStore((s) => s.selectedDistrict);
    43→  const selectedVenue = useMapStore((s) => s.selectedVenue);
    44→  const setSelectedState = useMapStore((s) => s.setSelectedState);
    45→  const setSelectedDistrict = useMapStore((s) => s.setSelectedDistrict);
    46→  const setSelectedVenue = useMapStore((s) => s.setSelectedVenue);
    47→  const sidebarOpen = useMapStore((s) => s.sidebarOpen);
    48→  const showWifiHeatmap = useMapStore((s) => s.showWifiHeatmap);
    49→  const toggleWifiHeatmap = useMapStore((s) => s.toggleWifiHeatmap);
    50→  const resetFilters = useMapStore((s) => s.resetFilters);
    51→
    52→  const [mapReady, setMapReady] = useState(false);
    53→  const [hoveredVenue, setHoveredVenue] = useState<LocationPin | null>(null);
    54→  const [tooltipPosition, setTooltipPosition] = useState<{ x: number; y: number } | null>(null);
    55→  const [legendCollapsed, setLegendCollapsed] = useState(false);
    56→
    57→  // Custom dark theme for the map (uses krackedmaps' actual CSS variable names)
    58→  const customTheme: Record<string, string> = {
    59→    "--sea": "#06080f",
    60→    "--grid": "rgba(224, 201, 127, 0.04)",
    61→    "--grid-size": "44px",
    62→    "--land": "#0f1830",
    63→    "--land-hover": "#1a2548",
    64→    "--land-active": "#e0c97f",
    65→    "--stroke": "rgba(224, 201, 127, 0.45)",
    66→    "--stroke-w": "1",
    67→    "--district": "#0d1b3a",
    68→    "--carve": "rgba(224, 201, 127, 0.35)",
    69→    "--carve-w": "0.6",
    70→    "--district-hi": "#e0c97f",
    71→    "--label": "#e0c97f",
    72→    "--label-dim": "rgba(224, 201, 127, 0.5)",
    73→    "--pin": "#e94560",
    74→    "--pin-ring": "rgba(233, 69, 96, 0.45)",
    75→    "--accent": "#e0c97f",
    76→    "--panel-bg": "#0d1b2a",
    77→    "--panel-edge": "rgba(224, 201, 127, 0.25)",
    78→  };
    79→
    80→  // Track whether we've drilled into a state (district view active)
    81→  // Derived from selectedState + drill event flag for more accuracy
    82→  const [drillEventFired, setDrillEventFired] = useState(false);
    83→
    84→  // Initialize map
    85→  useEffect(() => {
    86→    if (!mapRef.current) return;
    87→
    88→    const map = createMalaysiaMap(mapRef.current, {
    89→      theme: customTheme,
    90→      panel: false,
    91→      tooltip: true,
    92→      showDistricts: true,
    93→      interactive: true,
    94→      labels: false,
    95→    });
    96→
    97→    mapInstanceRef.current = map;
    98→    queueMicrotask(() => setMapReady(true));
    99→
   100→    // Handle state selection
   101→    map.on("select", (payload: StateSelectPayload) => {
   102→      if (payload.slug) {
   103→        setSelectedState(payload.slug);
   104→      } else {
   105→        setSelectedState(null);
   106→      }
   107→    });
   108→
   109→    // Handle drill into state (zoom to district view).
   110→    // krackedmaps emits `drill` with payload = state slug string (when drilling in)
   111→    // or null (when drilling out). We also support object payload for forward-compat.
   112→    map.on("drill", (payload: string | DrillPayload | null) => {
   113→      const stateSlug = typeof payload === "string"
   114→        ? payload
   115→        : (payload && (payload.state || payload.slug)) || null;
   116→      setDrillEventFired(!!stateSlug);
   117→      if (!stateSlug) {
   118→        // Drill-out event — clear district selection
   119→        setSelectedDistrict(null);
   120→      }
   121→    });
   122→
   123→    return () => {
   124→      if (map && typeof map.destroy === "function") {
   125→        map.destroy();
   126→      }
   127→    };
   128→  }, []);
   129→
   130→  // Build a lookup map from venue ID → venue object (stable, avoids stale closures)
   131→  const venueById = useMemo(() => {
   132→    const map = new Map<string, LocationPin>();
   133→    for (const loc of locations) {
   134→      map.set(loc.id, loc);
   135→    }
   136→    return map;
   137→  }, [locations]);
   138→
   139→  // Get locations for selected state (+ district)
   140→  const visibleLocations = useMemo(() => {
   141→    if (!selectedState) return locations;
   142→    const mappedState = SLUG_TO_STATE[selectedState] || selectedState;
   143→    return locations.filter((loc) => {
   144→      if (loc.state !== mappedState) return false;
   145→      if (selectedDistrict && loc.district !== selectedDistrict) return false;
   146→      return true;
   147→    });
   148→  }, [locations, selectedState, selectedDistrict]);
   149→
   150→  // Determine label visibility based on density and zoom level.
   151→  // - No state selected + few pins (<=4): show full names
   152→  // - Otherwise: HIDE labels — too dense or too zoomed in, use hover tooltips only
   153→  // This keeps the map clean and avoids overlap; the MapPinTooltip + bottom-left
   154→  // coordinate display provide context on hover.
   155→  const labelMode: 'full' | 'hidden' =
   156→    !selectedState && visibleLocations.length <= 4 ? 'full' : 'hidden';
   157→
   158→  // Label resolver based on mode
   159→  const getPinLabel = useCallback(
   160→    (loc: LocationPin): string => {
   161→      switch (labelMode) {
   162→        case 'full':
   163→          return loc.name.length <= 24 ? loc.name : loc.name.slice(0, 24) + '…';
   164→        case 'hidden':
   165→        default:
   166→          return ''; // No label — pins show on hover only
   167→      }
   168→    },
   169→    [labelMode],
   170→  );
   171→
   172→  // Update pins when locations or state changes
   173→  const updatePins = useCallback(() => {
   174→    if (!mapInstanceRef.current || !mapReady) return;
   175→    const map = mapInstanceRef.current;
   176→
   177→    // Clear existing pins
   178→    pinsRef.current.forEach((pinId) => {
   179→      try { map.removePin(pinId); } catch { /* ignore */ }
   180→    });
   181→    pinsRef.current.clear();
   182→
   183→    // Add filtered location pins with label optimization
   184→    for (const loc of visibleLocations) {
   185→      try {
   186→        const pinId = map.addPin({
   187→          lng: loc.longitude,
   188→          lat: loc.latitude,
   189→          label: getPinLabel(loc),
   190→          id: loc.id,
   191→        });
   192→        pinsRef.current.set(loc.id, pinId);
   193→      } catch {
   194→        // Skip pins outside bounds
   195→      }
   196→    }
   197→  }, [visibleLocations, mapReady, getPinLabel]);
   198→
   199→  useEffect(() => {
   200→    updatePins();
   201→  }, [updatePins]);
   202→
   203→  // Pin click handler — krackedmaps has no pinClick event, so we use DOM delegation
   204→  // Pins are rendered as <g class="pin" data-id="venueId"> in the SVG
   205→  useEffect(() => {
   206→    if (!mapInstanceRef.current || !mapReady) return;
   207→
   208→    const handlePinClick = (e: MouseEvent) => {
   209→      const pinEl = (e.target as Element).closest('.pin');
   210→      if (!pinEl) return;
   211→
   212→      const venueId = pinEl.getAttribute('data-id');
   213→      if (!venueId) return;
   214→
   215→      const venue = venueById.get(venueId);
   216→      if (venue) {
   217→        setSelectedVenue(venue);
   218→      }
   219→    };
   220→
   221→    mapInstanceRef.current.root.addEventListener('click', handlePinClick);
   222→    return () => {
   223→      mapInstanceRef.current?.root.removeEventListener('click', handlePinClick);
   224→    };
   225→  }, [mapReady, venueById, setSelectedVenue]);
   226→
   227→  // Derived: are we drilled into a state?
   228→  const isDrilledIn = !!selectedState && drillEventFired;
   229→
   230→  // Focus + drill into state when selected (district-level zoom)
   231→  useEffect(() => {
   232→    if (!mapInstanceRef.current) return;
   233→    const map = mapInstanceRef.current;
   234→    if (selectedState) {
   235→      try {
   236→        map.focus(selectedState);
   237→        // Drill into the state to reveal district boundaries
   238→        if (typeof map.drillInto === "function") {
   239→          map.drillInto(selectedState);
   240→        }
   241→      } catch {
   242→        // Focus might fail for states not in the main map
   243→      }
   244→    } else {
   245→      // Exit district view
   246→      try {
   247→        if (typeof map.drillInto === "function") {
   248→          map.drillInto(null);
   249→        }
   250→        map.focus(null);
   251→      } catch {
   252→        /* ignore */
   253→      }
   254→    }
   255→  }, [selectedState]);
   256→
   257→  // Highlight selected district on the map
   258→  useEffect(() => {
   259→    if (!mapInstanceRef.current) return;
   260→    const map = mapInstanceRef.current;
   261→    try {
   262→      if (selectedDistrict && typeof map.selectDistrict === "function") {
   263→        map.selectDistrict(selectedDistrict);
   264→      } else if (typeof map.selectDistrict === "function") {
   265→        // Deselect district when null — call with empty string or skip
   266→        // krackedmaps selectDistrict(key) — passing empty string clears highlight
   267→        map.selectDistrict("");
   268→      }
   269→    } catch {
   270→      /* ignore */
   271→    }
   272→  }, [selectedDistrict, isDrilledIn]);
   273→
   274→  // Pin hover tooltip — event delegation on SVG .pin[data-id] groups
   275→  useEffect(() => {
   276→    if (!mapInstanceRef.current || !mapReady) return;
   277→
   278→    const map = mapInstanceRef.current;
   279→
   280→    const handleMouseMove = (e: MouseEvent) => {
   281→      const pinEl = (e.target as Element).closest('.pin[data-id]');
   282→      if (!pinEl) return;
   283→
   284→      const venueId = pinEl.getAttribute('data-id');
   285→      if (!venueId) return;
   286→
   287→      const venue = venueById.get(venueId);
   288→      if (venue) {
   289→        setHoveredVenue(venue);
   290→        setTooltipPosition({ x: e.clientX, y: e.clientY });
   291→      }
   292→    };
   293→
   294→    const handleMouseLeave = () => {
   295→      setHoveredVenue(null);
   296→      setTooltipPosition(null);
   297→    };
   298→
   299→    map.root.addEventListener('mousemove', handleMouseMove);
   300→    map.root.addEventListener('mouseleave', handleMouseLeave);
   301→    return () => {
   302→      map.root.removeEventListener('mousemove', handleMouseMove);
   303→      map.root.removeEventListener('mouseleave', handleMouseLeave);
   304→    };
   305→  }, [mapReady, venueById]);
   306→
   307→  // Venue count for selected state
   308→  const venueCount = visibleLocations.length;
   309→
   310→  // Wi-Fi speed by state for heatmap
   311→  const wifiSpeedByState = useMemo(() => {
   312→    const stateMap = new Map<string, { total: number; count: number }>();
   313→    for (const loc of locations) {
   314→      const existing = stateMap.get(loc.state) || { total: 0, count: 0 };
   315→      stateMap.set(loc.state, { total: existing.total + loc.avgDownloadMbps, count: existing.count + 1 });
   316→    }
   317→    return Array.from(stateMap.entries())
   318→      .map(([state, { total, count }]) => ({
   319→        state,
   320→        avgSpeed: Math.round(total / count),
   321→        count,
   322→      }))
   323→      .sort((a, b) => b.avgSpeed - a.avgSpeed);
   324→  }, [locations]);
   325→
   326→  return (
   327→    <div className="relative w-full h-full min-h-screen bg-[#0a0a0f] overflow-hidden">
   328→      <StarfieldBackground />
   329→      <div ref={mapRef} className="w-full h-full min-h-screen" />
   330→
   331→      {/* Vignette effect around map edges */}
   332→      <div className="vignette-overlay" />
   333→
   334→      {/* Map pin hover tooltip — hidden when venue drawer is open */}
   335→      <MapPinTooltip venue={selectedVenue ? null : hoveredVenue} position={selectedVenue ? null : tooltipPosition} />
   336→
   337→      {/* District panel — shown when state is selected */}
   338→      <DistrictPanel />
   339→
   340→      {/* Drill controls — visible only when drilled in */}
   341→      <AnimatePresence>
   342→        {isDrilledIn && selectedState && (
   343→          <motion.div
   344→            initial={{ opacity: 0, y: -8, scale: 0.95 }}
   345→            animate={{ opacity: 1, y: 0, scale: 1 }}
   346→            exit={{ opacity: 0, y: -8, scale: 0.95 }}
   347→            transition={{ type: "spring", damping: 25, stiffness: 300 }}
   348→            className="absolute top-20 left-1/2 -translate-x-1/2 z-30 pointer-events-auto"
   349→            style={{ marginLeft: "130px" }}
   350→          >
   351→            <div className="flex items-center gap-1.5 bg-[#0d1b2a]/85 backdrop-blur-md border border-[#e0c97f]/30 rounded-full p-1 shadow-lg state-badge-glow">
   352→              <button
   353→                onClick={() => {
   354→                  setSelectedDistrict(null);
   355→                  setSelectedState(null);
   356→                }}
   357→                aria-label="Exit district zoom"
   358→                className="flex items-center gap-1.5 px-3 py-1.5 rounded-full text-[11px] font-medium text-[#e0c97f]/70 hover:text-[#e0c97f] hover:bg-[#e0c97f]/10 transition-all"
   359→              >
   360→                <ZoomOut className="w-3.5 h-3.5" />
   361→                Exit zoom
   362→              </button>
   363→              {selectedDistrict && (
   364→                <>
   365→                  <div className="w-px h-4 bg-[#e0c97f]/15" />
   366→                  <button
   367→                    onClick={() => setSelectedDistrict(null)}
   368→                    aria-label="Show all districts"
   369→                    className="flex items-center gap-1.5 px-3 py-1.5 rounded-full text-[11px] font-medium text-[#e0c97f]/70 hover:text-[#e0c97f] hover:bg-[#e0c97f]/10 transition-all"
   370→                  >
   371→                    <Crosshair className="w-3.5 h-3.5" />
   372→                    All districts
   373→                  </button>
   374→                </>
   375→              )}
   376→              <div className="w-px h-4 bg-[#e0c97f]/15" />
   377→              <div className="flex items-center gap-1.5 px-3 py-1.5 text-[11px] text-[#e0c97f]/40">
   378→                <ZoomIn className="w-3.5 h-3.5" />
   379→                <span className="font-mono">
   380→                  {selectedDistrict || "District view"}
   381→                </span>
   382→              </div>
   383→            </div>
   384→          </motion.div>
   385→        )}
   386→      </AnimatePresence>
   387→
   388→      {/* Map overlay gradient for readability */}
   389→      <div className="absolute inset-0 pointer-events-none">
   390→        <div className="absolute top-0 left-0 w-full h-32 bg-gradient-to-b from-[#0a0a0f]/80 to-transparent" />
   391→        <div className="absolute bottom-0 left-0 w-full h-48 bg-gradient-to-t from-[#0a0a0f]/90 to-transparent" />
   392→      </div>
   393→
   394→      {/* Weather Widget */}
   395→      <WeatherWidget />
   396→
   397→      {/* Quick Stats Overlay */}
   398→      <QuickStatsOverlay
   399→        visibleLocations={visibleLocations}
   400→        selectedState={selectedState}
   401→        sidebarOpen={sidebarOpen}
   402→      />
   403→
   404→      {/* Floating legend — glass-card enhanced with hover animations + collapse toggle */}
   405→      <div className="absolute top-20 right-4 glass-card pointer-events-auto min-w-[140px] overflow-hidden">
   406→        <div className="flex items-center justify-between px-4 pt-3.5 pb-2">
   407→          <p className="text-[10px] text-[#e0c97f]/50 font-semibold uppercase tracking-widest">Legend</p>
   408→          <button
   409→            onClick={() => setLegendCollapsed((v) => !v)}
   410→            aria-label={legendCollapsed ? "Expand legend" : "Collapse legend"}
   411→            title={legendCollapsed ? "Expand" : "Collapse"}
   412→            className="w-5 h-5 rounded flex items-center justify-center text-[#e0c97f]/40 hover:text-[#e0c97f] hover:bg-[#e0c97f]/8 transition-colors"
   413→          >
   414→            <motion.div animate={{ rotate: legendCollapsed ? -90 : 0 }} transition={{ duration: 0.2 }}>
   415→              <ChevronDown className="w-3 h-3" />
   416→            </motion.div>
   417→          </button>
   418→        </div>
   419→        <AnimatePresence initial={false}>
   420→          {!legendCollapsed && (
   421→            <motion.div
   422→              initial={{ height: 0, opacity: 0 }}
   423→              animate={{ height: 'auto', opacity: 1 }}
   424→              exit={{ height: 0, opacity: 0 }}
   425→              transition={{ duration: 0.25, ease: [0.22, 1, 0.36, 1] }}
   426→              className="overflow-hidden"
   427→            >
   428→              <div className="px-4 pb-4">
   429→                <div className="space-y-2">
   430→                  {(Object.entries(CATEGORY_CONFIG) as [string, { label: string; color: string; emoji: string }][]).map(([key, cfg]) => (
   431→                    <div
   432→                      key={key}
   433→                      className="flex items-center gap-2.5 min-w-0 legend-item-hover cursor-default"
   434→                      style={{ '--legend-color': cfg.color + '30' } as React.CSSProperties}
   435→                    >
   436→                      <div
   437→                        className="w-5 h-5 rounded flex-shrink-0 flex items-center justify-center text-xs legend-icon"
   438→                        style={{ backgroundColor: cfg.color + '20' }}
   439→                      >
   440→                        {cfg.emoji}
   441→                      </div>
   442→                      <span className="text-[11px] text-[#e0c97f]/70 whitespace-nowrap truncate">{cfg.label}</span>
   443→                    </div>
   444→                  ))}
   445→                </div>
   446→
   447→                {/* Animated venue count in view */}
   448→                <div className="mt-3 pt-3 border-t border-[#e0c97f]/10">
   449→                  <p className="text-[10px] text-[#e0c97f]/30 whitespace-nowrap">
   450→                    <AnimatePresence mode="popLayout">
   451→                      <motion.span
   452→                        key={venueCount}
   453→                        initial={{ opacity: 0, y: -6 }}
   454→                        animate={{ opacity: 1, y: 0 }}
   455→                        exit={{ opacity: 0, y: 6 }}
   456→                        transition={{ type: 'spring', damping: 20, stiffness: 300 }}
   457→                        className="inline-block font-semibold text-[#e0c97f]/50"
   458→                      >
   459→                        {venueCount}
   460→                      </motion.span>
   461→                    </AnimatePresence>{" "}
   462→                    venue{venueCount !== 1 ? "s" : ""} in view
   463→                  </p>
   464→                </div>
   465→
   466→                {/* Wi-Fi Heatmap Toggle */}
   467→                <div className="mt-3 pt-3 border-t border-[#e0c97f]/10">
   468→                  <button
   469→                    onClick={toggleWifiHeatmap}
   470→                    className={cn(
   471→                      "flex items-center gap-2 w-full px-2.5 py-2 rounded-lg text-[11px] font-medium transition-all duration-300 border",
   472→                      showWifiHeatmap
   473→                        ? "bg-[#22c55e]/12 border-[#22c55e]/30 text-[#22c55e]"
   474→                        : "bg-transparent border-[#e0c97f]/8 text-[#e0c97f]/35 hover:text-[#e0c97f]/60 hover:border-[#e0c97f]/15"
   475→                    )}
   476→                  >
   477→                    {showWifiHeatmap ? (
   478→                      <Wifi className="w-3.5 h-3.5" />
   479→            ) : (
   480→              <Activity className="w-3.5 h-3.5" />
   481→            )}
   482→            {showWifiHeatmap ? "Heatmap On" : "Wi-Fi Heatmap"}
   483→                  </button>
   484→                </div>
   485→              </div>
   486→            </motion.div>
   487→          )}
   488→        </AnimatePresence>
   489→      </div>
   490→
   491→      {/* Wi-Fi Speed Heatmap Overlay */}
   492→      <AnimatePresence>
   493→        {showWifiHeatmap && (
   494→          <motion.div
   495→            initial={{ opacity: 0, scale: 0.95 }}
   496→            animate={{ opacity: 1, scale: 1 }}
   497→            exit={{ opacity: 0, scale: 0.95 }}
   498→            transition={{ duration: 0.3 }}
   499→            className="absolute bottom-52 right-4 glass-card p-4 pointer-events-auto min-w-[200px] max-w-[240px]"
   500→          >
   501→            <div className="flex items-center gap-2 mb-3">
   502→              <Wifi className="w-4 h-4 text-[#22c55e]" />
   503→              <p className="text-[11px] text-[#e0c97f]/60 font-semibold">Avg Wi-Fi by State</p>
   504→            </div>
   505→            <div className="space-y-1.5">
   506→              {wifiSpeedByState.map(({ state, avgSpeed, count }) => (
   507→                <div key={state} className="flex items-center gap-2">
   508→                  <span className="text-[10px] text-[#e0c97f]/50 w-12 truncate">{state.replace('Kuala Lumpur', 'KL')}</span>
   509→                  <div className="flex-1 h-2 rounded-full bg-[#e0c97f]/8 overflow-hidden">
   510→                    <div
   511→                      className="h-full rounded-full transition-all duration-500"
   512→                      style={{
   513→                        width: `${Math.min(100, (avgSpeed / 300) * 100)}%`,
   514→                        background: avgSpeed > 100
   515→                          ? 'linear-gradient(90deg, #22c55e, #4ade80)'
   516→                          : avgSpeed > 50
   517→                            ? 'linear-gradient(90deg, #f59e0b, #fbbf24)'
   518→                            : 'linear-gradient(90deg, #ef4444, #f87171)',
   519→                      }}
   520→                    />
   521→                  </div>
   522→                  <span className="text-[10px] font-mono text-[#e0c97f]/60 w-16 text-right">{avgSpeed} <span className="text-[#e0c97f]/30">Mbps</span></span>
   523→                </div>
